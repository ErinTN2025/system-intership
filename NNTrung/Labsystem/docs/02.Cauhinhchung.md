- Set server giờ nội bộ: `sudo nano /etc/chrony/chrony.conf`
```bash
# Lấy Load balancer làm gốc
sudo tee /etc/chrony/chrony.conf > /dev/null <<'EOF'
pool ntp.ubuntu.com iburst maxsources 4
# Cho phép các node trong subnet đồng bộ về node này
allow 10.0.40.0/24
# Nếu mất internet vẫn tự phát giờ
local stratum 10
driftfile /var/lib/chrony/chrony.drift
makestep 1.0 3
rtcsync
EOF
sudo systemctl restart chrony

# Các Server còn lại trỏ đến Load balancer
sudo tee /etc/chrony/chrony.conf > /dev/null <<'EOF'
server 10.0.40.36 iburst prefer
pool ntp.ubuntu.com iburst maxsources 2
driftfile /var/lib/chrony/chrony.drift
makestep 1.0 3
rtcsync
EOF
sudo systemctl restart chrony
```

### 1.10 Cài đặt docker engine  
- Nên cài đặt phiên bản cụ thể của Docker để dễ quản lý và đồng bộ
- Thêm thư viện của Docker vào kho lưu trữ apt
```bash
# Thêm khóa GPG chính thức của Docker:
sudo apt update
sudo apt install ca-certificates curl
sudo install -m 0755 -d /etc/apt/keyrings
sudo curl -fsSL https://download.docker.com/linux/ubuntu/gpg -o /etc/apt/keyrings/docker.asc
sudo chmod a+r /etc/apt/keyrings/docker.asc

# Thêm kho lưu trữ vào Apt sources:
sudo tee /etc/apt/sources.list.d/docker.sources <<EOF
Types: deb
URIs: https://download.docker.com/linux/ubuntu
Suites: $(. /etc/os-release && echo "${UBUNTU_CODENAME:-$VERSION_CODENAME}")
Components: stable
Architectures: $(dpkg --print-architecture)
Signed-By: /etc/apt/keyrings/docker.asc
EOF

sudo apt update
```

### Cấu hình ufw (firewall) - Tương tự iptables
```bash

set -e
sudo ufw --force reset

sudo ufw default deny incoming
sudo ufw default deny outgoing
sudo ufw default deny routed


sudo ufw allow in on lo
sudo ufw allow out on lo

sudo ufw allow out 53          # DNS (tcp+udp)
sudo ufw allow out 80/tcp      # HTTP
sudo ufw allow out 443/tcp     # HTTPS
```

## Cấu hình Chrony - Sửa `etc/chrony/chrony.conf`
```bash
sudo cp /etc/chrony/chrony.conf /etc/chrony/chrony.conf.bak

sudo tee /etc/chrony/chrony.conf > /dev/null <<'EOF'
# Đồng bộ ra ngoài (chọn pool gần bạn, hoặc dùng Google/Cloudflare)
pool time.google.com iburst
pool 0.asia.pool.ntp.org iburst

# Cho phép các node trong dải cluster đồng bộ về CP1
allow 10.0.40.0/24

# Cho phép CP1 làm nguồn thời gian ngay cả khi tạm mất kết nối ra ngoài
# (tránh cả cụm mất đồng bộ nếu Internet rớt tạm thời)
local stratum 10

driftfile /var/lib/chrony/chrony.drift
rtcsync
makestep 1.0 3
EOF

sudo systemctl restart chrony
sudo systemctl enable chrony
```
- Check xem CP1 đã đồng bộ ra ngoài chưa:
```bash
chronyc tracking
chronyc sources -v
```
- Cài đặt và cấu hình từng Worker (client, trỏ về CP1)
