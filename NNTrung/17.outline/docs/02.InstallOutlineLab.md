## 2. File Docker Compose
- Tham khảo từ : docker-compose.yml 
```yml
services:
  outline:
    image: my-outline:latest
    env_file: ./docker.env
    ports:
      - "3000:3000"
    volumes:
      - storage-data:/var/lib/outline/data
    depends_on:
      - postgres
      - redis

  redis:
    image: redis
    env_file: ./docker.env
    expose:
      - "6379"
    # volumes:
    #  - ./redis.conf:/redis.conf
    # command: ["redis-server", "/redis.conf"]
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 30s
      retries: 3

  postgres:
    image: postgres:18
    env_file: ./docker.env
    expose:
      - "5432"
    volumes:
      - database-data:/var/lib/postgresql
    healthcheck:
      test: ["CMD", "pg_isready", "-d", "outline", "-U", "user"]
      interval: 30s
      timeout: 20s
      retries: 3
    environment:
      POSTGRES_USER: 'user'
      POSTGRES_PASSWORD: 'pass'
      POSTGRES_DB: 'outline'

  minio:
    image: minio/minio:latest
    command: server /data --console-address ":9001"
    environment:
      MINIO_ROOT_USER: minioadmin
      MINIO_ROOT_PASSWORD: minioadmin
      MINIO_SERVER_URL: https://minio.outline.dev:9000
      MINIO_BROWSER_REDIRECT_URL: https://minio.outline.dev:9001
      # Browser uploads/downloads from Outline local UI (HTTPS)
      MINIO_API_CORS_ALLOW_ORIGIN: "https://local.outline.dev:3000,https://localhost:3000,https://vite.outline.dev:3001,*"
    ports:
      - "127.0.0.1:9000:9000"
      - "127.0.0.1:9001:9001"
    volumes:
      - minio-data:/data
      # mkcert TLS — enables HTTPS so browser (HTTPS Outline) can load images
      - ./server/config/certs/minio:/root/.minio/certs:ro
    healthcheck:
      test: ["CMD", "curl", "-fk", "https://localhost:9000/minio/health/live"]
      interval: 5s
      timeout: 5s
      retries: 10
      start_period: 10s

  # One-shot: create outline bucket (API is HTTPS inside the network too)
  minio-init:
    image: minio/mc:latest
    depends_on:
      - minio
    entrypoint: ["/bin/sh", "-c"]
    command:
      - |
        set -e
        until mc alias set local https://minio:9000 minioadmin minioadmin --insecure; do echo waiting for minio; sleep 1; done
        mc mb -p local/outline --insecure || true
        mc ls local/ --insecure
        echo minio-init done

  https-portal:
    image: steveltn/https-portal:latest
    env_file: ./docker.env
    ports:
      - '80:80'
      - '443:443'
    links:
      - outline
    restart: always
    volumes:
      - https-portal-data:/var/lib/https-portal
    healthcheck:
      test: ["CMD", "service", "nginx", "status"]
      interval: 30s
      timeout: 20s
      retries: 3
    environment:
      DOMAINS: 'docs.mycompany.com -> http://outline:3000'
      STAGE: 'production'
      WEBSOCKET: 'true'
      CLIENT_MAX_BODY_SIZE: '0'

volumes:
  minio-data:
  https-portal-data:
  storage-data:
  database-data:
```
- Create a docker-compose.yml file, an example configuration with all dependencies dockerized and environment variables kept in docker.env is as follows. If you are installing the enterprise edition the image name should be replaced with `outlinewiki/outline-enterprise`.
- **Note**: It is recommended to pin the version of all images rather than relying on the latest tag so that you can remain in control of upgrades, e.g. `image: outlinewiki/outline:1.2.0` , `image: postgres:18`.

- Running: `docker compose up -d --build`
## 3. Run từ code
### 3.1 Build image base (biên dịch từ code)
```bash
# Đứng ở thư mục gốc của repo outline-main (nơi có file Dockerfile.base)
cd outline-main

docker build \
  -t my-outline-base:latest \    # ĐÂY LÀ TÊN (tag) đặt cho image
  -f Dockerfile.base \            # chỉ định dùng file nào để build (vì thư mục có 2 file Dockerfile)
  .                                # dấu chấm cuối = "build context" (thư mục chứa code để Docker đọc)
```
```bash
docker build \
  -t my-outline-base:latest \
  -f Dockerfile.base \
  .
```
```bash
docker build \
  -t my-outline:latest \
  -f Dockerfile \
  --build-arg BASE_IMAGE=my-outline-base:latest \
  .
```
```bash
# 1. Bật Postgres + Redis nền
docker compose up -d --build
```


## 4. Updating
1. Update to the latest image
  - Use docker pull docker.getoutline.com/outlinewiki/outline:latest to download the image or
  - If you are using docker-compose with a fixed version number then you will need to update the docker-compose.yml file to point to the latest version.
2. Run the container

- Check the logs:
```bash
docker compose logs outline --tail=50 -f
```

## 4. Lưu ý khi dùng OATH - Google WorkSpace
- Google OAuth không cho phép dùng địa chỉ IP riêng (private IP) làm redirect URI, kể cả khi bạn khai báo đúng trên Console. Google chỉ chấp nhận localhost hoặc domain có tên (hostname), không chấp nhận raw IP như `10.0.10.29`.
